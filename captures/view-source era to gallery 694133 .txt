                } catch(e) {}

                        // Fail-open: do NOT cache false, keep detection active.
                        self._logError('1002');
                        callback(true);

                        self._logError('1003');
                        callback(true);

                    self._logError('1005');
                    callback(true);

                this._logError('1006');

        // --- Detection Logic ---

        /** Checks if an element is hidden (disconnected, zero size, display:none, visibility:hidden, opacity:0, clipped) */
        isBlocked(element) {
            if (!element) return true;
            try {
                if (!element.isConnected) return true;
                if (element.offsetHeight === 0 || element.offsetWidth === 0) return true;
                var style = getComputedStyle(element);
                return (
                    style.display === "none" ||
                    style.visibility === "hidden" ||
                    style.opacity === "0" ||
                    style.clipPath === "inset(50%)"
                );
            } catch (e) {
                return true;
            }
        },

        /** Network-level signal: the real banner image failed to load / was hidden */
        isOriginalBannerBlocked() {
            var slot = window.__adpBannerSlot;
            if (window.__adpOriginalBlocked === true) return true;
            if (!slot || !slot.isConnected) return true;
            if (this.isBlocked(slot)) return true;
            var img = slot.querySelector("img");
            if (!img) return true;
            if (this.isBlocked(img)) return true;
            try {
                if (img.complete && img.naturalWidth === 0) return true;
            } catch (e) {}
            return false;
        },

        checkState() {
            // If already detected, stop checking
            if (this.state.warningActive) return;
            this.createBaits();
            var detected = false;

            // Check A0: Original banner image blocked (network-level)
            if (this.isOriginalBannerBlocked()) detected = true;

            // Check A: Generic bait visibility
            if (!detected && this.isBlocked(this.state.elements.baitGeneric)) detected = true;

            // Check A2: AdSense bait visibility
            if (!detected && this.isBlocked(this.state.elements.baitAdsense)) detected = true;

            // Check B: AdSense class manipulation
            if (!detected) {
                var allAds = document.querySelectorAll("ins." + this.config.classes.baitAdsense);
                for (var i = 0; i < allAds.length; i++) {
                    var ad = allAds[i];
                    var classes = Array.from(ad.classList);
                    var suspicious = classes.some(function(cls) {
                        return cls !== this.config.classes.baitAdsense &&
                               !cls.startsWith(this.config.classes.baitAdsensePermittedPrefix);
                    }.bind(this));
                    if (suspicious || ad.hasAttribute("title")) { detected = true; break; }
                }
            }

            // Check C: User-defined ad containers
            if (!detected && this.config.ids.watchedAdContainers && Array.isArray(this.config.ids.watchedAdContainers)) {
                for (var j = 0; j < this.config.ids.watchedAdContainers.length; j++) {
                    var el = document.getElementById(this.config.ids.watchedAdContainers[j]);
                    if (!el || this.isBlocked(el)) { detected = true; break; }
                }
            }

            if (detected) this.onAdBlockDetected();
        },

            this.checkState();
            var self = this;

        startSlotMonitor() {

        /**
         * Iframe Monitor (MutationObserver)
         * Watches for forced dimensions (1px !important) on ad iframes
         */
        startIframeMonitor() {
            var RULES = [
                /height\s*:\s*1px\s*!important/i,
                /width\s*:\s*1px\s*!important/i,
                /max-height\s*:\s*1px\s*!important/i,
                /max-width\s*:\s*1px\s*!important/i
            ];
            var iframeDetected = false;
            var self = this;
            var check = function() {
                if (iframeDetected || self.state.warningActive) return;
                var iframes = document.querySelectorAll('ins iframe[id^="aswift_"]');
                for (var i = 0; i < iframes.length; i++) {
                    var style = iframes[i].getAttribute('style');
                    if (!style) continue;
                    var count = 0;
                    for (var r = 0; r < RULES.length; r++) {
                        if (RULES[r].test(style)) count++;
                    }
                    if (count >= 2) {
                        iframeDetected = true;
                        self.onAdBlockDetected();
                        try { observer.disconnect(); } catch (e) {}
                        break;
                    }
                }
            };
            var observer = new MutationObserver(function(mutations) {
                var shouldCheck = false;
                for (var i = 0; i < mutations.length; i++) {
                    if (mutations[i].type === 'childList') shouldCheck = true;
                    else if (mutations[i].type === 'attributes' &&
                        (mutations[i].target.tagName === 'IFRAME' || mutations[i].target.tagName === 'INS')) shouldCheck = true;
                }
                if (shouldCheck) check();
            });
            try {
                observer.observe(document.documentElement, {
                    childList: true, subtree: true,
                    attributes: true, attributeFilter: ['style']
                });
            } catch (e) {}
            setInterval(check, 3500);
        },

         * checkFilterList() — NOT USED

         * Returns array of { selector, found, element }

        checkFilterList() {

            if (!url) return;

                        var el = document.querySelector(line);

                    } catch(e) {
                        // invalid selector, skip
                    }
                }

            };
            xhr.send();
        },

        }
    };
